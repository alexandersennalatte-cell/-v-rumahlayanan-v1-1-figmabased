 130|   function highlight(root, term) {
 131|     if (!term) return;
 132|     var escaped = term.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
 133|     var walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT, {
 134|       acceptNode: function (node) {
 135|         if (!node.nodeValue.trim()) return NodeFilter.FILTER_REJECT;
 136|         var p = node.parentNode.nodeName;
 137|         if (p === 'MARK' || p === 'SCRIPT' || p === 'STYLE') return NodeFilter.FILTER_REJECT;
 138|         return node.nodeValue.toLowerCase().indexOf(term) !== -1
 139|           ? NodeFilter.FILTER_ACCEPT
 140|           : NodeFilter.FILTER_REJECT;
 141|       }
 142|     });
 143|     var nodes = [];
 144|     var n;
 145|     while ((n = walker.nextNode())) nodes.push(n);
 146| 
 147|     nodes.forEach(function (node) {
 148|       var parts = node.nodeValue.split(new RegExp('(' + escaped + ')', 'ig'));
 149|       var frag = document.createDocumentFragment();
 150|       parts.forEach(function (part) {
 151|         if (part.toLowerCase() === term) {
 152|           var m = document.createElement('mark');
 153|           m.className = 'hl';
 154|           m.textContent = part;
 155|           frag.appendChild(m);
 156|         } else {
 157|           frag.appendChild(document.createTextNode(part));
 158|         }
 159|       });
 160|       node.parentNode.replaceChild(frag, node);
 161|     });
 162|   }

 164|   function runSearch(rawTerm) {

 174|       card.classList.toggle('is-match', hit);
 175|       card.classList.toggle('is-dimmed', !hit);
 176|       if (hit) matches.push(card);
 177|     });

 180|       var hit = normalize(li.textContent).indexOf(term) !== -1;
 181|       li.classList.toggle('is-match', hit);

 183|       if (hit) {

 186|         if (card) card.classList.add('is-match');
 187|       }

 191|       var hit = normalize(li.textContent).indexOf(term) !== -1;
